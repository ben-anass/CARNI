pub async fn create_document(
    State(db): State<DbState>,
    Json(mut payload): Json<CreateDocumentRequest>,
) -> Result<(StatusCode, Json<MedicalDocument>), (StatusCode, String)> {
    if payload.title.trim().is_empty() || payload.doctor_or_lab.trim().is_empty() {
        return Err((
            StatusCode::BAD_REQUEST,
            "Titre et émetteur du document obligatoires".to_string(),
        ));
    }
    // Validate file size and MIME type
    let max_size: u64 = 10 * 1024 * 1024; // 10MB
    if payload.file_size_bytes > max_size {
        return Err((
            StatusCode::BAD_REQUEST,
            "Fichier trop volumineux. Maximum 10 Mo".to_string(),
        ));
    }
    let allowed_types = ["application/pdf", "image/jpeg", "image/png"];
    if !allowed_types.contains(&payload.file_type.as_str()) {
        return Err((
            StatusCode::BAD_REQUEST,
            "Format non autorisé. Formats acceptés : PDF, JPG, PNG".to_string(),
        ));
    }

    // Sanitize filename to prevent path traversal or injection
    let clean_filename = std::path::Path::new(&payload.file_name)
        .file_name()
        .and_then(|n| n.to_str())
        .unwrap_or("document.pdf")
        .replace(['/', '\\', '\0'], "_");
    payload.file_name = clean_filename;

    let conn = db.lock().map_err(|_| {
        (
            StatusCode::INTERNAL_SERVER_ERROR,
            "DB Lock error".to_string(),
        )
    })?;
    match db::create_document(&conn, payload) {
        Ok(doc) => Ok((StatusCode::CREATED, Json(doc))),
        Err(e) => Err((StatusCode::INTERNAL_SERVER_ERROR, e.to_string())),
    }
}

pub async fn access_shared_session(
    State(db): State<DbState>,
    Path(token): Path<String>,
) -> Result<Json<DoctorAccessResponse>, (StatusCode, String)> {
    let conn = db.lock().map_err(|_| {
        (
            StatusCode::INTERNAL_SERVER_ERROR,
            "DB Lock error".to_string(),
        )
    })?;

    let session = match db::get_session_by_token(&conn, &token) {
        Ok(Some(s)) => s,
        Ok(None) => {
            return Err((
                StatusCode::NOT_FOUND,
                "Session de partage introuvable".to_string(),
            ))
        }
        Err(e) => return Err((StatusCode::INTERNAL_SERVER_ERROR, e.to_string())),
    };

    // Verify revoked
    if session.is_revoked {
        let _ = db::add_audit_log(
            &conn,
            "SESSION_ACCESS_DENIED",
            &session.recipient_label,
            "MEDECIN",
            "127.0.0.1",
            &format!("Tentative d'accès rejetée : jeton révoqué ({})", token),
            "REVOKED",
        );
        return Err((
            StatusCode::FORBIDDEN,
            "Cette session a été révoquée par le patient".to_string(),
        ));
    }

    // Verify expired
    if let Ok(exp) = chrono::DateTime::parse_from_rfc3339(&session.expires_at) {
        if exp < Utc::now() {
            let _ = db::add_audit_log(
                &conn,
                "SESSION_ACCESS_DENIED",
                &session.recipient_label,
                "MEDECIN",
                "127.0.0.1",
                &format!("Tentative d'accès rejetée : session expirée ({})", token),
                "WARNING",
            );
            return Err((
                StatusCode::GONE,
                "La durée de validité de cette session est expirée".to_string(),
            ));
        }
    }

    // Record access in database and audit log
    let _ = db::record_doctor_access(&conn, &session, "127.0.0.1");

    // Fetch full patient profile and sanitize according to patient's explicit consent scope
    let full_patient =
        db::get_patient(&conn).map_err(|e| (StatusCode::INTERNAL_SERVER_ERROR, e.to_string()))?;
    let consented_patient = db::filter_patient_by_scope(&full_patient, &session.scope);

    // Filter events according to scope
    let all_events = db::get_events(&conn).unwrap_or_default();
    let events = if session.scope.all_events {
        all_events
    } else {
        all_events
            .into_iter()
            .filter(|e| session.scope.event_ids.contains(&e.id))
            .collect()
    };

    // Filter documents according to scope
    let all_docs = db::get_documents(&conn).unwrap_or_default();
    let documents = if session.scope.all_docs {
        all_docs
    } else {
        all_docs
            .into_iter()
            .filter(|d| session.scope.doc_ids.contains(&d.id))
            .collect()
    };

    Ok(Json(DoctorAccessResponse {
        session,
        patient: consented_patient,
        events,
        documents,
    }))
}

use axum::http::Method;
use rusqlite::Connection;
use std::net::SocketAddr;
use std::sync::{Arc, Mutex};
use tower_http::cors::{Any, CorsLayer};

mod handlers;
mod models;
mod routes;
mod services;

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    println!("🇲🇦 Initialisation du serveur CARNI (Sovereign Health Journal MVP)...");

    // Initialize SQLite database
    let db_path = "carni.db";
    let conn = Connection::open(db_path)?;
    services::db::init_db(&conn)?;
    println!(
        "✓ Base de données SQLite initialisée et prête : {}",
        db_path
    );

    let state = Arc::new(Mutex::new(conn));

    // Configure CORS for Vite dev server & production
    let cors = CorsLayer::new()
        .allow_origin(Any)
        .allow_methods([
            Method::GET,
            Method::POST,
            Method::PUT,
            Method::DELETE,
            Method::OPTIONS,
        ])
        .allow_headers(Any);

    let app = routes::create_router(state).layer(cors);

    // Bind strictly to 127.0.0.1 for local security requirements
    let addr = SocketAddr::from(([127, 0, 0, 1], 3000));
    println!("✓ Serveur CARNI API démarré sur http://{}", addr);

    let listener = tokio::net::TcpListener::bind(addr).await?;
    axum::serve(listener, app).await?;

    Ok(())
}

use crate::handlers::*;
use axum::{
    routing::{delete, get, post, put},
    Router,
};

pub fn create_router(state: DbState) -> Router {
    Router::new()
        .route("/api/health", get(health_check))
        .route("/api/patient", get(get_patient).put(update_patient))
        .route("/api/events", get(get_events).post(create_event))
        .route("/api/events/{id}", put(update_event).delete(delete_event))
        .route("/api/documents", get(get_documents).post(create_document))
        .route("/api/documents/{id}", delete(delete_document))
        .route(
            "/api/shares",
            get(get_sharing_sessions).post(create_sharing_session),
        )
        .route("/api/shares/{id}/revoke", post(revoke_sharing_session))
        .route("/api/shares/access/{token}", get(access_shared_session))
        .route("/api/audit", get(get_audit_logs))
        .with_state(state)
}

use crate::models::*;

pub fn init_db(conn: &Connection) -> Result<()> {
    conn.execute_batch(
        "
        CREATE TABLE IF NOT EXISTS patient (
            id TEXT PRIMARY KEY,
            data_json TEXT NOT NULL
        );

        CREATE TABLE IF NOT EXISTS events (
            id TEXT PRIMARY KEY,
            title TEXT NOT NULL,
            event_date TEXT NOT NULL,
            category TEXT NOT NULL,
            doctor_or_facility TEXT NOT NULL,
            description TEXT NOT NULL,
            treatment_notes TEXT,
            vital_signs_json TEXT,
            attachments_json TEXT NOT NULL,
            created_at TEXT NOT NULL,
            updated_at TEXT
        );

        CREATE TABLE IF NOT EXISTS documents (
            id TEXT PRIMARY KEY,
            title TEXT NOT NULL,
            category TEXT NOT NULL,
            file_name TEXT NOT NULL,
            file_type TEXT NOT NULL,
            file_size_bytes INTEGER NOT NULL,
            issue_date TEXT NOT NULL,
            doctor_or_lab TEXT NOT NULL,
            notes TEXT,
            content_preview TEXT,
            tags_json TEXT NOT NULL,
            created_at TEXT NOT NULL
        );

        CREATE TABLE IF NOT EXISTS sharing_sessions (
            id TEXT PRIMARY KEY,
            token TEXT UNIQUE NOT NULL,
            recipient_label TEXT NOT NULL,
            purpose TEXT NOT NULL,
            scope_json TEXT NOT NULL,
            expires_at TEXT NOT NULL,
            created_at TEXT NOT NULL,
            is_revoked INTEGER NOT NULL DEFAULT 0,
            access_count INTEGER NOT NULL DEFAULT 0,
            last_accessed_at TEXT,
            one_time_use INTEGER NOT NULL DEFAULT 0
        );

        CREATE TABLE IF NOT EXISTS audit_logs (
            id TEXT PRIMARY KEY,
            timestamp TEXT NOT NULL,
            action TEXT NOT NULL,
            actor TEXT NOT NULL,
            actor_role TEXT NOT NULL,
            ip_address TEXT NOT NULL,
            details TEXT NOT NULL,
            status TEXT NOT NULL
        );
        ",
    )?;

    seed_if_empty(conn)?;
    Ok(())
}

pub fn get_events(conn: &Connection) -> Result<Vec<HealthEvent>> {
    let mut stmt = conn.prepare(
        "SELECT id, title, event_date, category, doctor_or_facility, description, treatment_notes, vital_signs_json, attachments_json, created_at, updated_at
         FROM events ORDER BY event_date DESC",
    )?;

    let rows = stmt.query_map([], |row| {

        let vital_signs: Option<EventVitalSigns> = vs_json

        Ok(HealthEvent {
            id: row.get(0)?,
            title: row.get(1)?,
            event_date: row.get(2)?,
            category: row.get(3)?,
            doctor_or_facility: row.get(4)?,
            description: row.get(5)?,
            treatment_notes: row.get(6)?,
            vital_signs,
            attachments,
            created_at: row.get(9)?,
            updated_at: row.get(10)?,
        })
    })?;

    let mut events = Vec::new();
    for r in rows {
        events.push(r?);
    }
    Ok(events)
}

pub fn get_sharing_sessions(conn: &Connection) -> Result<Vec<SharingSession>> {
    let mut stmt = conn.prepare(
        "SELECT id, token, recipient_label, purpose, scope_json, expires_at, created_at, is_revoked, access_count, last_accessed_at, one_time_use
         FROM sharing_sessions ORDER BY created_at DESC",
    )?;

    let rows = stmt.query_map([], |row| {
        let scope_json: String = row.get(4)?;
        let scope: SharingScope = serde_json::from_str(&scope_json).unwrap_or(SharingScope {
            all_events: false,
            event_ids: vec![],
            all_docs: false,
            doc_ids: vec![],
            include_allergies: true,
            include_vitals: true,
            include_emergency: true,
        });

        let is_revoked_i32: i32 = row.get(7)?;
        let one_time_i32: i32 = row.get(10)?;

        Ok(SharingSession {
            id: row.get(0)?,
            token: row.get(1)?,
            recipient_label: row.get(2)?,
            purpose: row.get(3)?,
            scope,
            expires_at: row.get(5)?,
            created_at: row.get(6)?,
            is_revoked: is_revoked_i32 != 0,
            access_count: row.get(8)?,
            last_accessed_at: row.get(9)?,
            one_time_use: one_time_i32 != 0,
        })
    })?;

    let mut sessions = Vec::new();
    for r in rows {
        sessions.push(r?);
    }
    Ok(sessions)
}

pub fn create_sharing_session(
    conn: &Connection,
    req: CreateShareRequest,
) -> Result<SharingSession> {
    let id = format!("sh_{}", Uuid::new_v4().simple());
    // 128-bit unguessable random token (32 hex characters)
    let token = format!("carni-sec-{}", Uuid::new_v4().simple());
    let now = Utc::now();
    let expires = (now + chrono::Duration::minutes(req.duration_minutes as i64)).to_rfc3339();
    let now_str = now.to_rfc3339();
    let scope_json = serde_json::to_string(&req.scope).unwrap_or_default();

    conn.execute(
        "INSERT INTO sharing_sessions (id, token, recipient_label, purpose, scope_json, expires_at, created_at, is_revoked, access_count, one_time_use)
         VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, 0, 0, ?8)",
        params![
            id,
            token,
            req.recipient_label,
            req.purpose,
            scope_json,
            expires,
            now_str,
            if req.one_time_use { 1 } else { 0 }
        ],
    )?;

    add_audit_log(
        conn,
        "SESSION_CREATED",
        "Yassine Mansouri (Patient)",
        "PATIENT",
        "127.0.0.1",
        &format!(
            "Création session QR Code (validité {} min) pour {}{}",
            req.duration_minutes,
            req.recipient_label,
            if req.one_time_use {
                " [Usage Unique]"
            } else {
                ""
            }
        ),
        "SUCCESS",
    )?;

    Ok(SharingSession {
        id,
        token,
        recipient_label: req.recipient_label,
        purpose: req.purpose,
        scope: req.scope,
        expires_at: expires,
        created_at: now_str,
        is_revoked: false,
        access_count: 0,
        last_accessed_at: None,
        one_time_use: req.one_time_use,
    })
}

         FROM sharing_sessions WHERE token = ?1",
    )?;

    let mut rows = stmt.query(params![token])?;
    if let Some(row) = rows.next()? {
        let scope_json: String = row.get(4)?;
        let scope: SharingScope = serde_json::from_str(&scope_json).unwrap_or(SharingScope {
            all_events: false,
            event_ids: vec![],
            all_docs: false,
            doc_ids: vec![],
            include_allergies: true,
            include_vitals: true,
            include_emergency: true,
        });

        let is_revoked_i32: i32 = row.get(7)?;
        let one_time_i32: i32 = row.get(10)?;

        Ok(Some(SharingSession {
            id: row.get(0)?,
            token: row.get(1)?,
            recipient_label: row.get(2)?,
            purpose: row.get(3)?,
            scope,
            expires_at: row.get(5)?,
            created_at: row.get(6)?,
            is_revoked: is_revoked_i32 != 0,
            access_count: row.get(8)?,
            last_accessed_at: row.get(9)?,
            one_time_use: one_time_i32 != 0,
        }))
    } else {
        Ok(None)
    }
}

pub fn record_doctor_access(
    conn: &Connection,
    session: &SharingSession,
    doctor_ip: &str,
) -> Result<()> {
    let now = Utc::now().to_rfc3339();
    let should_revoke = session.one_time_use;

    conn.execute(
        "UPDATE sharing_sessions SET access_count = access_count + 1, last_accessed_at = ?1, is_revoked = ?2 WHERE id = ?3",
        params![
            now,
            if should_revoke || session.is_revoked {
                1
            } else {
                0
            },
            session.id
        ],
    )?;

    add_audit_log(
        conn,
        "SESSION_ACCESSED",
        &session.recipient_label,
        "MEDECIN",
        doctor_ip,
        &format!(
            "Consultation autorisée du dossier médical (Session: {}){}",
            session.token,
            if should_revoke {
                " • Usage unique consommé"
            } else {
                ""
            }
        ),
        "SUCCESS",
    )?;

    Ok(())
}

pub fn get_audit_logs(conn: &Connection) -> Result<Vec<AuditLogEntry>> {
    let mut stmt = conn.prepare(
        "SELECT id, timestamp, action, actor, actor_role, ip_address, details, status
         FROM audit_logs ORDER BY timestamp DESC LIMIT 100",
    )?;

    let rows = stmt.query_map([], |row| {
        Ok(AuditLogEntry {
            id: row.get(0)?,
            timestamp: row.get(1)?,
            action: row.get(2)?,
            actor: row.get(3)?,
            actor_role: row.get(4)?,
            ip_address: row.get(5)?,
            details: row.get(6)?,
            status: row.get(7)?,
        })
    })?;

    let mut logs = Vec::new();
    for r in rows {
        logs.push(r?);
    }
    Ok(logs)
}

pub fn add_audit_log(
    conn: &Connection,
    action: &str,
    actor: &str,
    actor_role: &str,
    ip_address: &str,
    details: &str,
    status: &str,
) -> Result<()> {
    let id = format!("log_{}", Uuid::new_v4().simple());
    let now = Utc::now().to_rfc3339();

    conn.execute(
        "INSERT INTO audit_logs (id, timestamp, action, actor, actor_role, ip_address, details, status)
         VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8)",
        params![id, now, action, actor, actor_role, ip_address, details, status],
    )?;

    Ok(())
}
